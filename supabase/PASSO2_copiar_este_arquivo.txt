do $$ begin
  create type papel_usuario as enum ('ADMIN', 'SECRETARIA', 'PROFESSOR', 'ALUNO');
exception when duplicate_object then null; end $$;

do $$ begin
  create type turno_aula as enum ('MATUTINO', 'VESPERTINO', 'NOTURNO', 'SABADO', 'EAD');
exception when duplicate_object then null; end $$;

do $$ begin
  create type situacao_aluno as enum ('ATIVO', 'INATIVO', 'TRANCADO', 'CONCLUIDO', 'TRANSFERIDO');
exception when duplicate_object then null; end $$;

do $$ begin
  create type resultado_nota as enum ('APTO', 'NAO_APTO', 'F_NOTA', 'REP_FALTAS', 'PENDENTE', 'DISPENSADO', 'DESISTENTE');
exception when duplicate_object then null; end $$;

do $$ begin
  create type presenca_tipo as enum ('P', 'F');
exception when duplicate_object then null; end $$;

-- Perfil de cada pessoa. O 'id' é o mesmo do Supabase Auth (auth.users),
-- onde a senha fica guardada com hash — nunca em texto puro.
create table if not exists public.usuarios (
  id            uuid primary key references auth.users(id) on delete cascade,
  nome          text not null,
  email         text not null,
  login         text unique not null,   -- matrícula (aluno) ou usuário (professor/admin)
  papel         papel_usuario not null,
  ativo         boolean not null default true,
  trocar_senha  boolean not null default true,  -- força troca no primeiro acesso
  criado_em     timestamptz not null default now(),
  atualizado_em timestamptz not null default now()
);

create index if not exists idx_usuarios_login on public.usuarios(lower(login));
create index if not exists idx_usuarios_papel on public.usuarios(papel) where ativo;

-- Permissões finas por módulo (usado para o papel SECRETARIA / funcionários)
create table if not exists public.permissoes_modulo (
  usuario_id  uuid not null references public.usuarios(id) on delete cascade,
  modulo      text not null,
  ver         boolean not null default false,
  criar       boolean not null default false,
  editar      boolean not null default false,
  excluir     boolean not null default false,
  imprimir    boolean not null default false,
  exportar    boolean not null default false,
  primary key (usuario_id, modulo)
);

-- PARTE 10 — AUDITORIA (quem mudou o quê)

create table if not exists public.auditoria (
  id            bigserial primary key,
  usuario_id    uuid references public.usuarios(id) on delete set null,
  usuario_nome  text,
  acao          text not null,        -- INSERT | UPDATE | DELETE
  tabela        text not null,
  registro_id   text,
  valor_antes   jsonb,
  valor_depois  jsonb,
  criado_em     timestamptz not null default now()
);

create index if not exists idx_auditoria_tabela on public.auditoria(tabela, criado_em desc);

-- Gatilho genérico de auditoria
create or replace function public.registrar_auditoria()
returns trigger language plpgsql security definer set search_path = public as $$
declare v_nome text;
begin
  select nome into v_nome from public.usuarios where id = auth.uid();
  insert into public.auditoria (usuario_id, usuario_nome, acao, tabela, registro_id, valor_antes, valor_depois)
  values (
    auth.uid(), v_nome, tg_op, tg_table_name,
    coalesce((to_jsonb(new)->>'id'), (to_jsonb(old)->>'id')),
    case when tg_op in ('UPDATE','DELETE') then to_jsonb(old) end,
    case when tg_op in ('INSERT','UPDATE') then to_jsonb(new) end
  );
  return coalesce(new, old);
end $$;

-- Auditoria obrigatória em notas e frequência (proteção jurídica da escola)
drop trigger if exists trg_auditoria_notas on public.notas;
create trigger trg_auditoria_notas
  after insert or update or delete on public.notas
  for each row execute function public.registrar_auditoria();

drop trigger if exists trg_auditoria_frequencia on public.frequencia;
create trigger trg_auditoria_frequencia
  after insert or update or delete on public.frequencia
  for each row execute function public.registrar_auditoria();

drop trigger if exists trg_auditoria_alunos on public.alunos;
create trigger trg_auditoria_alunos
  after insert or update or delete on public.alunos
  for each row execute function public.registrar_auditoria();

-- Atualiza 'atualizado_em' automaticamente
create or replace function public.tocar_atualizado_em()
returns trigger language plpgsql as $$
begin new.atualizado_em := now(); return new; end $$;

do $$
declare t text;
begin
  foreach t in array array['usuarios','cursos','disciplinas','turmas','alunos','professores','notas','registros_modulo']
  loop
    execute format('drop trigger if exists trg_touch_%1$s on public.%1$s', t);
    execute format('create trigger trg_touch_%1$s before update on public.%1$s
                    for each row execute function public.tocar_atualizado_em()', t);
  end loop;
end $$;


-- ============================================================================
-- PARTE 10B — FUNÇÕES DE APOIO À SEGURANÇA
-- ============================================================================
-- SECURITY DEFINER: rodam com privilégio elevado para poder consultar a tabela
-- 'usuarios' sem cair em recursão infinita das próprias políticas de segurança.

create or replace function public.papel_atual()
returns papel_usuario
language sql stable security definer set search_path = public as $$
  select papel from public.usuarios where id = auth.uid() and ativo
$$;

create or replace function public.eh_admin()
returns boolean
language sql stable security definer set search_path = public as $$
  select coalesce(public.papel_atual() = 'ADMIN', false)
$$;

-- Gestão = administração + secretaria (quem enxerga a escola inteira)
create or replace function public.eh_gestao()
returns boolean
language sql stable security definer set search_path = public as $$
  select coalesce(public.papel_atual() in ('ADMIN', 'SECRETARIA'), false)
$$;

create or replace function public.eh_professor()
returns boolean
language sql stable security definer set search_path = public as $$
  select coalesce(public.papel_atual() = 'PROFESSOR', false)
$$;

create or replace function public.aluno_atual_id()
returns uuid
language sql stable security definer set search_path = public as $$
  select id from public.alunos where usuario_id = auth.uid()
$$;

create or replace function public.professor_atual_id()
returns uuid
language sql stable security definer set search_path = public as $$
  select id from public.professores where usuario_id = auth.uid()
$$;

-- O diário pertence ao professor logado?
create or replace function public.meu_diario(p_diario_id uuid)
returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.diarios d
    where d.id = p_diario_id
      and d.professor_id = public.professor_atual_id()
  )
$$;

-- O diário está aberto para lançamento? (bloqueio por prazo/fechamento)
create or replace function public.diario_aberto(p_diario_id uuid)
returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.diarios d
    where d.id = p_diario_id and d.fechado = false
  )
$$;

-- IMPORTANTE: as funções abaixo evitam RECURSÃO INFINITA entre políticas.
-- Uma política da tabela A não pode consultar a tabela B diretamente se a
-- política de B consultar A de volta. Por serem SECURITY DEFINER, estas
-- funções ignoram o RLS internamente e quebram esse ciclo.

create or replace function public.sou_professor_da_turma(p_turma_id uuid)
returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.diarios d
    where d.turma_id = p_turma_id and d.professor_id = public.professor_atual_id()
  )
$$;

create or replace function public.estou_matriculado(p_turma_id uuid)
returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.matriculas m
    where m.turma_id = p_turma_id and m.aluno_id = public.aluno_atual_id()
  )
$$;

create or replace function public.turma_do_diario(p_diario_id uuid)
returns uuid
language sql stable security definer set search_path = public as $$
  select turma_id from public.diarios where id = p_diario_id
$$;

create or replace function public.diario_da_aula(p_aula_id uuid)
returns uuid
language sql stable security definer set search_path = public as $$
  select diario_id from public.aulas where id = p_aula_id
$$;

-- Visitante sem login não executa nada disso; quem fez login, sim.
do $$
declare f text;
begin
  foreach f in array array[
    'papel_atual()', 'eh_admin()', 'eh_gestao()', 'eh_professor()',
    'aluno_atual_id()', 'professor_atual_id()',
    'meu_diario(uuid)', 'diario_aberto(uuid)',
    'sou_professor_da_turma(uuid)', 'estou_matriculado(uuid)',

  end loop;
end $$;

-- TRAVA CONTRA ESCALADA DE PRIVILÉGIO
-- A política acima deixa a pessoa editar a própria linha (nome, e-mail...).
-- Sem esta trava, um aluno poderia trocar o próprio 'papel' para ADMIN e
-- ganhar acesso a tudo. O gatilho abaixo impede isso no banco.
create or replace function public.protege_campos_criticos()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  if not public.eh_gestao() then
    if new.papel is distinct from old.papel then
      raise exception 'Não é permitido alterar o papel do usuário.';
    end if;
    if new.ativo is distinct from old.ativo then
      raise exception 'Não é permitido alterar o status de acesso.';
    end if;
    if new.login is distinct from old.login then
      raise exception 'Não é permitido alterar o login.';
    end if;
  end if;
  return new;
end $$;

drop trigger if exists trg_protege_usuarios on public.usuarios;
create trigger trg_protege_usuarios
  before update on public.usuarios
  for each row execute function public.protege_campos_criticos();

revoke all on all functions in schema public from anon;

  grant usage, select on sequences to authenticated;

  revoke all on tables from anon;
